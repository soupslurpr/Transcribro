cmake_minimum_required(VERSION 3.22)
project(transcribro_runtime_verification LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(RUNTIME_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../speech/src/main/cpp/medium")
find_package(Threads REQUIRED)
add_subdirectory("${RUNTIME_SOURCE}" runtime)
include_directories("${RUNTIME_SOURCE}")
if(DEFINED MOONSHINE_SOURCE)
  add_executable(run_stream run_stream.cpp)
  target_include_directories(run_stream PRIVATE "${MOONSHINE_SOURCE}")
  target_link_libraries(run_stream PRIVATE transcribro_standalone ${CMAKE_DL_LIBS})
endif()
if(TRANSCRIBRO_SANITIZERS)
  target_compile_options(transcribro_standalone PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(transcribro_standalone PRIVATE -fsanitize=address,undefined)
endif()
add_executable(run_components run_components.cpp)
target_link_libraries(run_components PRIVATE transcribro_standalone)
add_executable(test_runtime test_runtime.cpp)
target_link_libraries(test_runtime PRIVATE transcribro_standalone)
add_executable(benchmark_frontend benchmark_frontend.cpp)
target_link_libraries(benchmark_frontend PRIVATE transcribro_standalone)
add_executable(benchmark_loading benchmark_loading.cpp)
target_link_libraries(benchmark_loading PRIVATE transcribro_standalone)
add_executable(test_projection test_projection.cpp)
target_include_directories(test_projection PRIVATE "${ORT_SOURCE}/onnxruntime/core/mlas/inc")
target_link_libraries(test_projection PRIVATE onnxruntime_mlas ${CMAKE_DL_LIBS} m)
add_executable(benchmark_tiled benchmark_tiled.cpp)
add_executable(benchmark_head_layout benchmark_head_layout.cpp)
add_executable(benchmark_token_selection benchmark_token_selection.cpp)
add_executable(benchmark_attention benchmark_attention.cpp)
add_executable(benchmark_local_attention benchmark_local_attention.cpp)
foreach(executable benchmark_head_layout benchmark_token_selection benchmark_attention benchmark_local_attention)
  target_include_directories(${executable} PRIVATE "${ORT_SOURCE}/onnxruntime/core/mlas/inc" "${RUNTIME_SOURCE}")
  target_link_libraries(${executable} PRIVATE onnxruntime_mlas ${CMAKE_DL_LIBS} m)
endforeach()
target_link_libraries(benchmark_head_layout PRIVATE Threads::Threads)
foreach(executable test_projection benchmark_tiled benchmark_head_layout benchmark_token_selection benchmark_attention benchmark_local_attention)
  target_compile_options(${executable} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endforeach()
if(TRANSCRIBRO_TRACE)
  target_compile_definitions(run_components PRIVATE TRANSCRIBRO_TRACE)
endif()
if(DEFINED ORT_TEST_LIBRARY)
  add_executable(run_reference_components run_components.cpp)
  target_compile_definitions(run_reference_components PRIVATE TRANSCRIBRO_ORT_REFERENCE)
  target_include_directories(run_reference_components PRIVATE "${ORT_TEST_INCLUDE}")
  target_link_libraries(run_reference_components PRIVATE "${ORT_TEST_LIBRARY}")
  add_executable(verify_audio verify_audio.cpp)
  target_include_directories(verify_audio PRIVATE "${ORT_TEST_INCLUDE}")
  target_link_libraries(verify_audio PRIVATE transcribro_standalone "${ORT_TEST_LIBRARY}")
endif()
if(DEFINED ZSTD_TEST_LIBRARY)
  add_executable(benchmark_decompression benchmark_decompression.cpp)
  target_include_directories(benchmark_decompression PRIVATE "${ZSTD_TEST_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../speech/src/main/cpp")
  target_link_libraries(benchmark_decompression PRIVATE "${ZSTD_TEST_LIBRARY}")
  target_compile_options(benchmark_decompression PRIVATE -Wall -Wextra -Werror)
endif()
if(TRANSCRIBRO_SANITIZERS)
  foreach(executable run_components test_runtime run_stream verify_audio benchmark_frontend benchmark_loading benchmark_decompression test_projection benchmark_tiled benchmark_head_layout benchmark_token_selection benchmark_attention benchmark_local_attention)
    if(TARGET ${executable})
      target_compile_options(${executable} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${executable} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
