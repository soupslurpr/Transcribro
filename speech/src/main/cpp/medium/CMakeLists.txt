cmake_minimum_required(VERSION 3.22)
project(transcribro_medium LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TRANSCRIBRO_SANITIZERS "Instrument native verification" OFF)
option(TRANSCRIBRO_TRACE "Enable verification tensor callbacks" OFF)
option(TRANSCRIBRO_PROFILE "Enable verification component timing" OFF)
option(TRANSCRIBRO_EMBED_RUNTIME "Link the runtime into JNI" OFF)
if(TRANSCRIBRO_EMBED_RUNTIME)
  add_library(transcribro_standalone STATIC medium.cpp stream.cpp)
else()
  add_library(transcribro_standalone SHARED medium.cpp stream.cpp)
  set_target_properties(transcribro_standalone PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()
target_include_directories(transcribro_standalone PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(transcribro_standalone PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
foreach(diagnostic TRACE PROFILE)
  if(TRANSCRIBRO_${diagnostic})
    target_compile_definitions(transcribro_standalone PRIVATE TRANSCRIBRO_${diagnostic})
  endif()
endforeach()
if(NOT EXISTS "${EIGEN_SOURCE}/Eigen/Core")
  message(FATAL_ERROR "Prepare the pinned native inputs with :speech:prepareNativeSources")
endif()
include(mlas.cmake)
target_include_directories(transcribro_standalone PRIVATE "${ORT_SOURCE}/onnxruntime/core/mlas/inc")
target_include_directories(transcribro_standalone SYSTEM PRIVATE "${EIGEN_SOURCE}")
target_compile_definitions(transcribro_standalone PRIVATE EIGEN_MPL2_ONLY)
target_link_libraries(transcribro_standalone PRIVATE onnxruntime_mlas ${CMAKE_DL_LIBS} m Threads::Threads)
if(NOT TRANSCRIBRO_EMBED_RUNTIME)
  target_link_options(transcribro_standalone PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" -Wl,--exclude-libs,ALL)
  set_property(TARGET transcribro_standalone APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
if(ANDROID)
  target_link_options(transcribro_standalone PRIVATE -Wl,-z,max-page-size=16384)
endif()
if(TRANSCRIBRO_SANITIZERS)
  target_compile_options(transcribro_standalone PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(transcribro_standalone PRIVATE -fsanitize=address,undefined)
endif()
